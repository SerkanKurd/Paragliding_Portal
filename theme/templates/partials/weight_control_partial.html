function calculateWeight() { // Inputlardan değerleri al const pWeight = parseFloat(document.getElementById('pilotWeight').value) || 0; const gWeight = parseFloat(document.getElementById('gearWeight').value) || 0; const minW = parseFloat(document.getElementById('minW').value); const maxW = parseFloat(document.getElementById('maxW').value); // Toplamı hesapla ve yazdır const total = pWeight + gWeight; document.getElementById('totalWeightDisplay').innerText = total.toFixed(1); // Yüzdeyi hesapla const range = maxW - minW; let percentage = ((total - minW) / range) * 100; // Barı hareket ettir (Yüzdeyi 0-100 arasında sınırla ama dışarı taşmasına izin ver) const pointer = document.getElementById('weightPointer'); const label = document.getElementById('percentageLabel'); pointer.style.left = percentage + "%"; label.innerText = "%" + Math.round(percentage); // Uyarı Sistemini Yönet const alertBox = document.getElementById('weightAlert'); const alertMsg = document.getElementById('alertMessage'); if (percentage > 100) { alertBox.classList.remove('hidden'); alertBox.querySelector('.alert').className = "alert alert-error shadow-lg"; alertMsg.innerText = "⚠️ Limit Üstündesin! Kanat uçuş karakteri sertleşecektir."; } else if (percentage < 0) { alertBox.classList.remove('hidden'); alertBox.querySelector('.alert').className = "alert alert-warning shadow-lg"; alertMsg.innerText = "⚠️ Limit Altındasın! Kanat kapanmalara karşı daha hassas olabilir."; } else { alertBox.classList.add('hidden'); } } // Sayfa yüklendiğinde ilk hesaplamayı yap window.onload = calculateWeight;
